# -Energy-Consumption-Forecasting-with-ARIMA
Energy Consumption Forecasting with ARIMA


Data: PJME Hourly Energy Consumption
# 📌 Objective: Forecast next 30 days using ARIMA
# 📌 EDA: Visual trend, seasonality decomposition
# 📌 Model: ARIMA with auto_arima tuning
# 📌 Evaluation: RMSE, MAE on last 30 days
# 📌 Tools: Python, Pandas, NumPy, Matplotlib, Seaborn, Statsmodels, Pmdarima


data: [https://www.kaggle.com/datasets/robikscube/hourly-energy-consumption?resource=download]
